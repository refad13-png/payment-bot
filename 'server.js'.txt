const path = require('path');
const crypto = require('crypto');

const PORT = process.env.PORT || 3000;
const WEB_APP_URL = process.env.RENDER_EXTERNAL_URL || 'http://localhost:3000';
const ADMIN_PASSWORD = 'admin123';

const app = express();

app.use(express.json());

app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'index.html')); });
app.get('/admin', (req, res) => { res.sendFile(path.join(__dirname, 'admin.html')); });

const products = [
  { id: 1, name: 'Basic', price: 990, period: 'mes', emoji: 'star' },
  { id: 2, name: 'Pro', price: 2490, period: 'mes', emoji: 'bolt' },
  { id: 3, name: 'Premium', price: 4990, period: 'mes', emoji: 'gem' },
];

const orders = [];

app.get('/api/products', (req, res) => { res.json(products); });

app.post('/api/pay', (req, res) => {
  const { productId, cardNumber, cardExpiry, cardCvc, cardHolder, email, telegramId } = req.body;
  const errors = [];
  const product = products.find(p => p.id === productId);

  if (!product) errors.push('Product not found');
  if (!cardNumber || !/^\d{16}$/.test(cardNumber.replace(/\s/g, ''))) errors.push('Card: 16 digits');
  if (!cardExpiry || !/^(0[1-9]|1[0-2])\/\d{2}$/.test(cardExpiry)) errors.push('Expiry: MM/YY');
  if (!cardCvc || !/^\d{3,4}$/.test(cardCvc)) errors.push('CVC: 3-4 digits');
  if (!cardHolder || cardHolder.trim().length < 3) errors.push('Name required');
  if (!email || !email.includes('@')) errors.push('Invalid email');

  if (errors.length > 0) return res.status(400).json({ success: false, errors });

  const transactionId = 'txn_' + crypto.randomBytes(6).toString('hex');
  const maskedCard = '**** ' + cardNumber.replace(/\s/g, '').slice(-4);

  const order = {
    id: orders.length + 1, transactionId, product: product.name,
    amount: product.price, currency: 'RUB', cardMasked: maskedCard,
    cardHolder, email, telegramId: telegramId || null,
    status: 'paid', createdAt: new Date().toISOString()
  };

  orders.push(order);

  setTimeout(() => {
    res.json({ success: true, transactionId, amount: product.price, productName: product.name, maskedCard, message: 'Payment successful' });
    if (telegramId) {
      bot.telegram.sendMessage(telegramId, 'Payment successful: ' + product.price + ' RUB\nProduct: ' + product.name + '\nCard: ' + maskedCard).catch(() => {});
    }
  }, 1500);
});

app.post('/api/admin/login', (req, res) => {
  if (req.body.password === ADMIN_PASSWORD) {
    res.json({ success: true, token: 'admin_token' });
  } else {
    res.status(401).json({ success: false });
  }
});

app.get('/api/admin/orders', (req, res) => {
  if (req.headers.authorization !== 'Bearer admin_token') {
    return res.status(401).json({ error: 'Unauthorized' });
  }
  res.json({
    totalOrders: orders.length,
    totalRevenue: orders.reduce((s, o) => s + o.amount, 0),
    orders: orders.slice().reverse()
  });
});

    }
  });
});

bot.action('about', async (ctx) => {

app.listen(PORT, () => console.log('Server: ' + PORT));
